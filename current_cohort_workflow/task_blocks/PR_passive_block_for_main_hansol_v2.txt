class TimedProgressiveRatioExperiment(ProgressiveRatioExperiment):
    """Same PR task, but the session stops by itself after SESSION_SECONDS
    (a reward already in progress always finishes) and the elapsed time is printed."""
    def make_trial(self, idx):
        trial = super().make_trial(idx)
        remaining = self.max_duration - (time.time() - self._start_time)
        trial.RESPONSE_WINDOW = max(0.0, remaining)
        return trial

    def _clock(self):
        el = int(time.time() - self._start_time) if self._start_time else 0
        left = max(0, int(self.max_duration) - el)
        return f"{el // 60:02d}:{el % 60:02d} elapsed / {left // 60:02d}:{left % 60:02d} left"

    def trial_summary(self, idx, outcome):
        print(f"[{self._clock()}] Trial {idx} summary: {outcome}")

    def run(self):
        done = threading.Event()
        def ticker():
            while not done.wait(PRINT_EVERY):
                if self._start_time:
                    print(f"[TIME] {self._clock()}")
        threading.Thread(target=ticker, daemon=True).start()
        try:
            super().run()
        finally:
            done.set()
            if self._start_time:
                print(f"[TIME] session finished: {self._clock()}")


def replay_clock(t0, total):
    el = int(time.time() - t0) if t0 else 0
    left = max(0, int(total) - el)
    return f"{el // 60:02d}:{el % 60:02d} elapsed / {left // 60:02d}:{left % 60:02d} left"


def check_replay_source(src, out):
    """Stop before anything starts when the log to replay is wrong. Returns (rewards, length in s) of the source."""
    if not os.path.isfile(src):
        raise SystemExit(f"[REPLAY] STOP: '{src}' not found in {os.getcwd()}")
    if os.path.abspath(src) == os.path.abspath(out):
        raise SystemExit("[REPLAY] STOP: OUTPUT is the same file as REPLAY_SOURCE")
    recs = []
    with open(src, encoding="utf-8") as f:
        for line in f:
            if line.strip():
                try:
                    recs.append(json.loads(line))
                except ValueError:
                    pass
    meta = recs[0] if recs and recs[0].get("tag") == "meta" else None
    if meta is None:
        raise SystemExit(f"[REPLAY] STOP: '{src}' has no meta line (not a task log?)")
    if "replay" in str(meta.get("paradigm", "")).lower():
        raise SystemExit(f"[REPLAY] STOP: '{src}' is itself a passive (replay) log - use the ACTIVE partner's log")
    n_rew = sum(r.get("tag") == "reward_cmd" for r in recs)
    length = float(recs[-1].get("elapsed", 0.0))
    started = datetime.fromisoformat(meta["t"])
    problems = []
    if not any(r.get("tag") == "experiment_end" for r in recs):
        problems.append("it has no experiment_end (the partner's session is still running, or it crashed)")
    if started.date() != datetime.now().date():
        problems.append(f"it is from {started.date()}, not today")
    print(f"[REPLAY] source {src}: started {started:%Y-%m-%d %H:%M}, {meta.get('paradigm')}, "
          f"{n_rew} rewards, {length:.0f} s")
    if problems and not REPLAY_ANYWAY:
        raise SystemExit("[REPLAY] STOP: " + "; ".join(problems) + " - fix REPLAY_SOURCE "
                         "(or set REPLAY_ANYWAY = True on purpose)")
    for p in problems:
        print(f"[REPLAY] WARNING (REPLAY_ANYWAY = True): {p}")
    return n_rew, length
